02循环结构

用 for 循环输出 1~20 的偶数，并用不同颜色渲染（类似彩虹）。

let chen_colors = ["#ef4444","#f97316","#eab308","#22c55e","#06b6d4","#3b82f6","#8b5cf6","#ec4899"];
for (let chen_i = 2; chen_i <= 20; chen_i += 2) {
    // 取余数作为颜色索引
    chen_paint(chen_i, chen_colors[chen_i % chen_colors.length]);
}

用 while 从 1 开始累加，直到总和超过 100 停止。

let chen_sum = 0, chen_n = 1;
while (chen_sum <= 100) {
    chen_sum += chen_n;
    chen_n++;
}
// 停在多少？和是多少？
等你运行…

用 do ... while 模拟摇骰子，直到摇到 6 点才停。

let chen_count = 0, chen_roll;
do {
    chen_roll = Math.floor(Math.random() * 6) + 1;
    chen_count++;
} while (chen_roll !== 6);
// do while 至少执行一次，即使初始条件不满足
等你运行…

for ... of 遍历数组元素；for ... in 遍历对象的键名。

let chen_names = ["小明", "小红", "小刚"];

// for ... of：遍历值
for (let chen_n of chen_names) {
    console.log(chen_n);
}

// for ... in：遍历键
let chen_student = { name: "陈弘毅", age: 18 };
for (let chen_k in chen_student) {
    console.log(chen_k, chen_student[chen_k]);
}
等你运行…
